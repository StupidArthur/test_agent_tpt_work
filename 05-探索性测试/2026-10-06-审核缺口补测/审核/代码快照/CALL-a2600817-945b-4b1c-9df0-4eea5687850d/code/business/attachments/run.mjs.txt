// G7-09/G7-11/G7-12: attachment container scroll, clear entry, sent-bubble fixed attachment.
import path from 'node:path';
import { withApp, sleep, newTask, selectProject, typeAndSend, waitTerminal } from '../../automation/tpt.mjs';

async function clearAttachments(page) {
  for (let i = 0; i < 60; i++) {
    const b = page.locator('[role="group"][aria-label="待发送附件"] button[aria-label^="移除文件"]').first();
    if (!(await b.count())) break;
    await b.click({ timeout: 3000 }).catch(() => {});
    await sleep(100);
  }
}

export async function attachmentsRun(ctx, args) {
  const root = ctx.taskRoot;
  const run = ctx.runtime.extension_run;
  const countDir = path.join(root, '夹具', '本轮', run, 'attachments', 'count');
  const files30 = [];
  for (let i = 1; i <= 30; i++) files30.push(path.join(countDir, `count-${String(i).padStart(2, '0')}.txt`));
  const files3 = [
    path.join(root, '夹具', '本轮', run, 'attachments', 'attachment-a.txt'),
    path.join(root, '夹具', '本轮', run, 'attachments', 'attachment-b.txt'),
    path.join(root, '夹具', '本轮', run, 'attachments', 'attachment-c.md'),
  ];
  return withApp(ctx, async (page) => {
    const { recorder } = ctx;
    const out = {};
    await newTask(page);
    await selectProject(page).catch(() => {});
    await clearAttachments(page);

    // ---- G7-09 30 attachments scroll ----
    const a30 = await recorder.action('添加30份附件', 'setInputFiles', 'count-01..30.txt', async () => {
      await page.locator('input[type="file"]').first().setInputFiles(files30);
      await sleep(2500);
    });
    const rScroll = await recorder.read('附件容器可滚动', 'draft.attachments.container', {
      channel: 'derived', scope: '当前草稿附件容器', locator: '[role="group"][aria-label="待发送附件"]',
    }, async () => {
      const raw = await page.evaluate(() => {
        const rail = document.querySelector('[role="group"][aria-label="待发送附件"]');
        if (!rail) return null;
        const cs = getComputedStyle(rail);
        return {
          scrollWidth: rail.scrollWidth, clientWidth: rail.clientWidth,
          scrollHeight: rail.scrollHeight, clientHeight: rail.clientHeight,
          overflowX: cs.overflowX, overflowY: cs.overflowY,
          itemCount: rail.querySelectorAll('[class*="_item"]').length,
        };
      });
      if (!raw) return { value: null, raw: null, reason: 'attachment rail not found' };
      const xScrollable = raw.scrollWidth > raw.clientWidth && /auto|scroll/.test(raw.overflowX);
      const yScrollable = raw.scrollHeight > raw.clientHeight && /auto|scroll/.test(raw.overflowY);
      let axis = null;
      if (xScrollable) { await page.evaluate(() => { const r = document.querySelector('[role="group"][aria-label="待发送附件"]'); r.scrollLeft = r.scrollWidth; }); axis = 'x'; }
      else if (yScrollable) { await page.evaluate(() => { const r = document.querySelector('[role="group"][aria-label="待发送附件"]'); r.scrollTop = r.scrollHeight; }); axis = 'y'; }
      await sleep(600);
      return { value: xScrollable || yScrollable, raw: { ...raw, axis }, derivation: '(scrollWidth>clientWidth&&overflowX auto|scroll) || (scrollHeight>clientHeight&&overflowY auto|scroll)' };
    });
    const rLast = await recorder.read('最后附件可见', 'draft.attachments.last', {
      channel: 'dom', scope: '当前草稿附件容器/count-30.txt', locator: '[role="group"][aria-label="待发送附件"]',
    }, async () => {
      const v = await page.evaluate(() => {
        const rail = document.querySelector('[role="group"][aria-label="待发送附件"]');
        if (!rail) return null;
        const leaf = [...rail.querySelectorAll('*')].filter((e) => e.children.length === 0 && /count-30\.txt/.test(e.textContent || ''));
        if (!leaf.length) return { visible: false, found: false };
        const el = leaf[0];
        const r = el.getBoundingClientRect();
        const rr = rail.getBoundingClientRect();
        const visible = r.left >= rr.left - 2 && r.right <= rr.right + 2 && r.width > 0;
        return { visible, found: true, rect: { left: r.left, right: r.right }, rail: { left: rr.left, right: rr.right } };
      });
      return { value: v ? v.visible : null, raw: v, reason: v === null ? 'rail not found' : (v && !v.found ? 'count-30.txt node not found' : undefined) };
    });
    out.G7_09 = { action: a30.event_id, scroll: rScroll.event_id, scrollValue: rScroll.value, last: rLast.event_id, lastValue: rLast.value };

    // ---- G7-11 clear entry with 3 attachments ----
    await clearAttachments(page);
    const a3 = await recorder.action('添加3份附件', 'setInputFiles', 'attachment-a/b/c', async () => {
      await page.locator('input[type="file"]').first().setInputFiles(files3);
      await sleep(2000);
    });
    const rClearEntry = await recorder.read('附件清空入口', 'draft.attachments.clear-entry', {
      channel: 'dom', scope: '当前草稿附件工具区', locator: 'button in attachments dock',
    }, async () => {
      const raw = await page.evaluate(() => {
        const dock = document.querySelector('[data-slot="conversation.input.attachments"]') || document.querySelector('[role="group"][aria-label="待发送附件"]');
        if (!dock) return null;
        const btns = [...dock.querySelectorAll('button')].map((b) => ({ aria: b.getAttribute('aria-label'), text: (b.innerText || '').replace(/\s+/g, ' ').trim() })).filter((b) => b.aria || b.text);
        const clear = btns.some((b) => /清空|全部移除|清除|删除全部|移除全部/.test((b.aria || '') + b.text));
        return { btns, clear };
      });
      if (!raw) return { value: null, raw: null, reason: 'attachments dock not found' };
      return { value: raw.clear, raw, derivation: 'a batch-clear control present in attachments dock' };
    });
    let clearClicked = false; let afterCount = null;
    if (rClearEntry.value === true) {
      const aClick = await recorder.action('点击清空附件', 'click', '批量清空入口', async () => {
        clearClicked = await page.evaluate(() => {
          const dock = document.querySelector('[data-slot="conversation.input.attachments"]') || document.querySelector('[role="group"][aria-label="待发送附件"]');
          const b = dock && [...dock.querySelectorAll('button')].find((x) => /清空|全部移除|清除|删除全部|移除全部/.test((x.getAttribute('aria-label') || '') + (x.innerText || '')));
          if (b) { b.click(); return true; } return false;
        });
        await sleep(1200);
      });
      out.G7_11_clearAction = aClick.event_id;
      const rAfter = await recorder.read('清空后的草稿附件数', 'draft.attachments.after-clear', {
        channel: 'dom', scope: '当前草稿附件容器', locator: '[role="group"][aria-label="待发送附件"] [class*="_item"]',
      }, async () => {
        const n = await page.evaluate(() => { const rail = document.querySelector('[role="group"][aria-label="待发送附件"]'); return rail ? rail.querySelectorAll('[class*="_item"]').length : null; });
        return { value: n, raw: { count: n }, reason: n === null ? 'rail not found' : undefined };
      });
      afterCount = rAfter.value;
      out.G7_11_afterRead = rAfter.event_id;
    }
    out.G7_11 = { action: a3.event_id, entryRead: rClearEntry.event_id, entryValue: rClearEntry.value, clearClicked, afterCount };
    return out;
  });
}
