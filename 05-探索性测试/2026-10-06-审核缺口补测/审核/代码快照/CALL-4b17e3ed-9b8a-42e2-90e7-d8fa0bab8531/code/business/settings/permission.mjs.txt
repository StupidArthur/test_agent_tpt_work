// G6-06: default permission setting -> new task permission, and restore.
import { withApp, sleep, newTask } from '../../automation/tpt.mjs';
import { isSettingsOpen, openSettings, openSection, closeSettings, readSelectorValue, setSelector } from '../../automation/settings.mjs';

async function readComposerPermission(page) {
  return page.evaluate(() => {
    const el = document.querySelector('[data-slot="conversation.input.permission"]');
    if (!el) return null;
    const t = (el.innerText || '').replace(/\s+/g, ' ').trim();
    return { text: t, session: document.querySelector('[data-conversation-session]')?.getAttribute('data-conversation-session') ?? null };
  });
}

export async function settingsPermission(ctx, args) {
  return withApp(ctx, async (page) => {
    const { recorder } = ctx;
    const out = {};
    await openSettings(page);
    await openSection(page, '常规');

    const rInitial = await recorder.read('常规设置默认权限', 'settings.permission', {
      channel: 'dom', scope: '设置/常规/权限行', locator: '权限 selector',
    }, async () => {
      const v = await readSelectorValue(page, '权限');
      return { value: v, raw: { selector_text: v } };
    });
    out.initial = rInitial.value;

    const actSet = await recorder.action('设为工作区内修改', 'click+menuitem', '权限=工作区内修改', async () => {
      await setSelector(page, '权限', '工作区内修改');
    });
    await closeSettings(page);
    await openSettings(page);
    await openSection(page, '常规');
    const rReopen = await recorder.read('重开后默认权限设置', 'settings.permission.reopen', {
      channel: 'dom', scope: '设置/常规/权限行', locator: '权限 selector',
    }, async () => {
      const v = await readSelectorValue(page, '权限');
      return { value: v, raw: { selector_text: v } };
    });
    out.reopen = rReopen.value;
    await closeSettings(page);

    const actNew1 = await recorder.action('创建改后新任务', 'click', '新建任务', async () => { await newTask(page); });
    const rNew1 = await recorder.read('改后新空白任务权限', 'newtask.permission.changed', {
      channel: 'dom', scope: '主任务视图/composer权限控件', locator: '[data-slot="conversation.input.permission"]',
    }, async () => {
      const r = await readComposerPermission(page);
      return { value: r ? r.text : null, raw: r, reason: r ? undefined : 'permission control not found' };
    });
    out.new1 = rNew1.value;

    // restore initial
    await openSettings(page);
    await openSection(page, '常规');
    const actRestore = await recorder.action('恢复默认权限初值', 'click+menuitem', '权限=' + rInitial.value, async () => {
      await setSelector(page, '权限', rInitial.value);
    });
    await closeSettings(page);
    await openSettings(page);
    await openSection(page, '常规');
    const rRestored = await recorder.read('恢复后默认权限设置', 'settings.permission.restored', {
      channel: 'dom', scope: '设置/常规/权限行', locator: '权限 selector',
    }, async () => {
      const v = await readSelectorValue(page, '权限');
      return { value: v, raw: { selector_text: v } };
    });
    out.restored = rRestored.value;
    await closeSettings(page);

    const actNew2 = await recorder.action('创建恢复后新任务', 'click', '新建任务', async () => { await newTask(page); });
    const rNew2 = await recorder.read('恢复后另一个新任务权限', 'newtask.permission.restored', {
      channel: 'dom', scope: '主任务视图/composer权限控件', locator: '[data-slot="conversation.input.permission"]',
    }, async () => {
      const r = await readComposerPermission(page);
      return { value: r ? r.text : null, raw: r, reason: r ? undefined : 'permission control not found' };
    });
    out.new2 = rNew2.value;

    out.reads = { initial: rInitial.event_id, reopen: rReopen.event_id, new1: rNew1.event_id, restored: rRestored.event_id, new2: rNew2.event_id };
    out.actions = { set: actSet.event_id, new1: actNew1.event_id, restore: actRestore.event_id, new2: actNew2.event_id };
    return out;
  });
}
