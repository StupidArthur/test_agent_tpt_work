const fs=require('fs'),path=require('path'),crypto=require('crypto'),root=path.resolve(__dirname,'..'),cases=JSON.parse(fs.readFileSync(path.join(root,'用例/cases.json'),'utf8')),issues=[],counts={通过:0,失败:0,不确定:0,未执行:0};
for(const c of cases){const f=path.join(root,'结果',c.id+'.json');if(!fs.existsSync(f)){counts.未执行++;continue;}try{const r=JSON.parse(fs.readFileSync(f,'utf8'));if(r.case_id!==c.id)issues.push(c.id+': case_id mismatch');if(!['通过','失败','不确定'].includes(r.status)){issues.push(c.id+': forbidden or invalid status');continue;}counts[r.status]++;for(const key of ['started_at','ended_at','observed','cleanup'])if(!r[key])issues.push(c.id+': missing '+key);if(!Array.isArray(r.actual_steps)||!r.actual_steps.length)issues.push(c.id+': missing actual steps');if(!Array.isArray(r.evidence)||!r.evidence.length)issues.push(c.id+': missing evidence');else for(const e of r.evidence){const p=path.resolve(root,e.path||'');if(!p.startsWith(root+path.sep)){issues.push(c.id+': unsafe evidence path');continue;}if(!fs.existsSync(p)){issues.push(c.id+': evidence missing');continue;}const h=crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');if(h!==e.sha256)issues.push(c.id+': evidence hash mismatch');}}catch(e){issues.push(c.id+': '+e.message);}}
console.log(JSON.stringify({total:cases.length,counts,complete:counts.未执行===0&&issues.length===0,issues,note:'Structural validation does not approve business conclusions'},null,2));process.exitCode=issues.length?1:0;
