// 业务函数层：composer 附件（添加/读取/移除/清空/预览/滚动）。
import { connect } from '../../automation/session.mjs';
import { SEL } from '../../automation/conversation.mjs';

const AREA = '[data-slot="conversation.input.attachments"]';
const CARD = `${AREA} [title]`;
const NAME = `${AREA} [class*="name"]`;

async function ensureAddMenu(page) {
  if (!(await page.locator('input[type="file"][multiple]').count())) {
    await page.locator('button[aria-label="添加文件或调用指令"]').first().click();
    await page.waitForTimeout(1000);
  }
}

// 经 添加本地文件 一次选择多个文件。
export async function addLocalFiles(ctx, args = {}) {
  const { paths } = args;
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    const act = await rec.action('添加本地文件', 'setInputFiles', { count: paths.length }, async () => {
      await page.locator('button[aria-label="添加文件或调用指令"]').first().click();
      await page.waitForTimeout(900);
      const menu = page.getByText('添加本地文件', { exact: true }).first();
      if (await menu.count()) { await menu.click(); await page.waitForTimeout(700); }
      await ensureAddMenu(page);
      await page.locator('input[type="file"][multiple]').first().setInputFiles(paths);
      await page.waitForTimeout(1800);
    });
    const names = await rec.read('当前草稿附件名顺序', 'composer-attachment-names', { channel: 'dom', scope: 'composer 附件区', locator: NAME }, async () => {
      const els = page.locator(`${AREA} .gSkjMW_name`);
      const n = await els.count();
      const list = [];
      for (let i = 0; i < n; i++) list.push((await els.nth(i).innerText()).trim());
      return { value: list, raw: { count: n, names: list } };
    });
    return { action_refs: [act.event_id], observations: { names } };
  } finally { await conn.close(); }
}

// 读取当前草稿附件数、名列表、数量提示文本与区域文本。
export async function readDraftState(ctx) {
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    const names = await rec.read('当前草稿附件名列表', 'composer-attachment-names', { channel: 'dom', scope: 'composer 附件区', locator: NAME }, async () => {
      const els = page.locator(`${AREA} .gSkjMW_name`);
      const n = await els.count();
      const list = [];
      for (let i = 0; i < n; i++) list.push((await els.nth(i).innerText()).trim());
      return { value: list, raw: { count: n, names: list } };
    });
    const count = await rec.read('当前草稿附件数', 'composer-attachment-count', { channel: 'dom', scope: 'composer 附件区', locator: NAME }, async () => {
      const n = await page.locator(`${AREA} .gSkjMW_name`).count();
      return { value: n, raw: { count: n } };
    });
    const hint = await rec.read('附件数量提示', 'composer-attachment-hint', { channel: 'dom', scope: 'composer 附件区', locator: AREA }, async () => {
      const text = (await page.locator(AREA).first().innerText()).replace(/\s+/g, ' ').trim();
      const m = /(\d+)\s*\/\s*(\d+)/.exec(text);
      const value = m ? `${m[1]}/${m[2]}` : null;
      return { value, raw: { text, matched: value } };
    });
    return { observations: { names, count, hint } };
  } finally { await conn.close(); }
}

// 读取附件区域可见的“清空”入口是否存在，并读取当前数量；可选点击。
export async function clearAttachments(ctx, args = {}) {
  const { click = false } = args;
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    const exists = await rec.read('附件清空入口存在', 'composer-attachment-clear-entry', { channel: 'dom', scope: 'composer 附件区', locator: AREA }, async () => {
      const cands = page.locator(`${AREA} button`);
      const n = await cands.count();
      const labels = [];
      for (let i = 0; i < n; i++) { const t = ((await cands.nth(i).innerText()) || '') + ' ' + ((await cands.nth(i).getAttribute('aria-label')) || ''); labels.push(t.trim()); }
      const hasClear = labels.some((t) => /清空|全部移除|清除全部/.test(t));
      return { value: hasClear, raw: { buttonLabels: labels }, derivation: '/清空|全部移除|清除全部/.test(labels)' };
    });
    if (click && exists.value) {
      await rec.action('点击清空附件', 'click', null, async () => {
        const cands = page.locator(`${AREA} button`);
        const n = await cands.count();
        for (let i = 0; i < n; i++) {
          const t = ((await cands.nth(i).innerText()) || '') + ' ' + ((await cands.nth(i).getAttribute('aria-label')) || '');
          if (/清空|全部移除|清除全部/.test(t)) { await cands.nth(i).click(); break; }
        }
        await page.waitForTimeout(1200);
      });
    }
    const count = await rec.read('草稿附件数', 'composer-attachment-count', { channel: 'dom', scope: 'composer 附件区', locator: NAME }, async () => {
      const n = await page.locator(`${AREA} .gSkjMW_name`).count();
      return { value: n, raw: { count: n } };
    });
    return { observations: { exists, count } };
  } finally { await conn.close(); }
}

// 读取指定附件卡内图像加载状态（img.complete && naturalWidth>0）。
export async function readImageState(ctx, args = {}) {
  const { name } = args;
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    const img = await rec.read('附件图像加载', 'attachment-image', { channel: 'dom', scope: 'composer 附件卡', locator: `${AREA} img` }, async () => {
      const el = page.locator(`${AREA} [title="${name}"] img`).first();
      const n = await el.count();
      let raw = { count: n, complete: null, naturalWidth: null };
      if (n) raw = await el.evaluate((im) => ({ count: 1, complete: im.complete, naturalWidth: im.naturalWidth }));
      const value = n > 0 && raw.complete === true && raw.naturalWidth > 0;
      return { value, raw, derivation: 'count>0 && complete && naturalWidth>0' };
    });
    return { observations: { img } };
  } finally { await conn.close(); }
}

// 打开/关闭图片整屏预览。
export async function togglePreview(ctx, args = {}) {
  const { name, open = true } = args;
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    if (open) {
      await rec.action('打开图片预览', 'click', { name }, async () => {
        await page.locator(`${AREA} [title="${name}"] [class*="thumb"], ${AREA} [title="${name}"] img`).first().click();
        await page.waitForTimeout(1500);
      });
    } else {
      await rec.action('关闭图片预览', 'escape', null, async () => {
        await page.keyboard.press('Escape');
        await page.waitForTimeout(1200);
      });
    }
    const layer = await rec.read('图片预览层可见', 'attachment-preview-layer', { channel: 'dom', scope: '应用主表面/图片预览', locator: '[class*="preview"], [role="dialog"]' }, async () => {
      const visible = await page.evaluate(() => {
        const vis = (el) => !!(el.offsetWidth || el.offsetHeight || el.getClientRects().length);
        const cands = [...document.querySelectorAll('[class*="preview"],[role="dialog"],[class*="lightbox"],[class*="viewer"]')].filter(vis);
        let imgLoaded = false;
        for (const c of cands) { for (const im of c.querySelectorAll('img')) { if (im.complete && im.naturalWidth > 0) imgLoaded = true; } }
        return { layers: cands.length, imgLoaded };
      });
      const value = open ? (visible.layers > 0 && visible.imgLoaded) : (visible.layers === 0);
      return { value, raw: visible, derivation: open ? 'layers>0 && imgLoaded' : 'layers===0' };
    });
    return { observations: { layer } };
  } finally { await conn.close(); }
}

// 读取附件容器可滚动信息并滚到末端，返回末附件可见状态。
export async function scrollAttachments(ctx, args = {}) {
  const { lastName } = args;
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    const scroll = await rec.read('附件容器可滚动', 'attachment-container-scroll', { channel: 'dom', scope: 'composer 附件容器', locator: `${AREA} [role="group"]` }, async () => {
      const info = await page.evaluate(() => {
        const el = document.querySelector('[data-slot="conversation.input.attachments"] [role="group"]') || document.querySelector('[data-slot="conversation.input.attachments"]');
        if (!el) return { found: false };
        const cs = getComputedStyle(el);
        const horiz = el.scrollWidth > el.clientWidth && /auto|scroll/.test(cs.overflowX);
        const vert = el.scrollHeight > el.clientHeight && /auto|scroll/.test(cs.overflowY);
        el.scrollLeft = el.scrollWidth; el.scrollTop = el.scrollHeight;
        return { found: true, scrollWidth: el.scrollWidth, clientWidth: el.clientWidth, scrollHeight: el.scrollHeight, clientHeight: el.clientHeight, overflowX: cs.overflowX, overflowY: cs.overflowY, horiz, vert };
      });
      const value = info.found ? (info.horiz || info.vert) : null;
      return { value, raw: info, derivation: 'horiz || vert' };
    });
    await page.waitForTimeout(600);
    const visible = await rec.read('末附件可见', 'attachment-last-visible', { channel: 'dom', scope: 'composer 附件卡', locator: `${AREA} [title="${lastName}"]` }, async () => {
      const el = page.locator(`${AREA} [title="${lastName}"]`).first();
      const n = await el.count();
      const vis = n ? await el.isVisible() : false;
      return { value: vis, raw: { count: n, visible: vis }, derivation: 'count>0 && visible' };
    });
    return { observations: { scroll, visible } };
  } finally { await conn.close(); }
}

// 读取附件相关的可见提示/反馈（超额、拒绝等）。
export async function readFeedback(ctx) {
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    const fb = await rec.read('附件提示反馈', 'attachment-feedback', { channel: 'dom', scope: '应用主表面/提示', locator: '[role="alert"],[class*="toast"],[class*="message"],[class*="toast"]' }, async () => {
      const text = await page.evaluate(() => {
        const vis = (el) => !!(el.offsetWidth || el.offsetHeight || el.getClientRects().length);
        const cands = [...document.querySelectorAll('[role="alert"],[class*="toast"],[class*="Toast"],[class*="message"],[class*="tip"]')].filter(vis);
        return cands.map((c) => (c.innerText || '').replace(/\s+/g, ' ').trim()).filter(Boolean);
      });
      const joined = text.join(' | ') || null;
      return { value: joined, raw: { candidates: text } };
    });
    return { observations: { fb } };
  } finally { await conn.close(); }
}

// 读取指定附件卡完整文本，并判断是否含肯定解析声明。
export async function readCardText(ctx, args = {}) {
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    const text = await rec.read('未知格式卡片状态文本', 'attachment-card-text', { channel: 'dom', scope: 'composer 附件卡', locator: `${AREA} [title="${args.name}"]` }, async () => {
      const el = page.locator(`${AREA} [title="${args.name}"]`).first();
      const n = await el.count();
      const t = n ? (await el.innerText()).replace(/\s+/g, ' ').trim() : null;
      const claimed = t ? /(解析成功|内容已识别|已解析|识别成功)/.test(t) : false;
      return { value: n > 0 ? t : null, raw: { count: n, text: t, positiveClaim: claimed } };
    });
    return { observations: { text } };
  } finally { await conn.close(); }
}

// 悬停指定附件卡并读取 title/tooltip 完整名称。
export async function readCardTitle(ctx, args = {}) {
  const { name } = args;
  const conn = await connect(ctx);
  try {
    const { page } = conn;
    const rec = ctx.recorder;
    const act = await rec.action('悬停附件卡', 'hover', { name }, async () => {
      await page.locator(`${AREA} [title="${name}"]`).first().hover();
      await page.waitForTimeout(1200);
    });
    const title = await rec.read('长文件名完整提示', 'attachment-card-title', { channel: 'dom', scope: 'composer 附件卡', locator: `${AREA} [title="${name}"]` }, async () => {
      const el = page.locator(`${AREA} [title="${name}"]`).first();
      const n = await el.count();
      const attr = n ? await el.getAttribute('title') : null;
      const tooltip = await page.evaluate(() => {
        const vis = (e) => !!(e.offsetWidth || e.offsetHeight || e.getClientRects().length);
        const t = [...document.querySelectorAll('[role="tooltip"],[class*="tooltip"]')].filter(vis).map((x) => (x.innerText || '').trim()).filter(Boolean);
        return t;
      });
      const value = attr || tooltip[0] || null;
      return { value, raw: { count: n, titleAttr: attr, tooltips: tooltip } };
    });
    return { action_refs: [act.event_id], observations: { title } };
  } finally { await conn.close(); }
}
