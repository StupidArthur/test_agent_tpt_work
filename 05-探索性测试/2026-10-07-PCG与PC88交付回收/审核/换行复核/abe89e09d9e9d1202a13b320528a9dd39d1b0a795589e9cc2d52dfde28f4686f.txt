// Thin tool-call bridge. No case scheduling, UI implementation or test verdicts.
import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import {fileURLToPath, pathToFileURL} from 'node:url';
import {createRequire} from 'node:module';
const root=path.resolve(fileURLToPath(new URL('../',import.meta.url)));
const parse=p=>JSON.parse(fs.readFileSync(p,'utf8'));
const hash=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const relative=p=>path.relative(root,p).replaceAll('\\','/');
function file(p){const result=fs.realpathSync(path.resolve(root,p));if(!result.startsWith(root+path.sep)||!fs.statSync(result).isFile())throw Error('File must be inside this task');return result;}
const catalog=parse(path.join(root,'code/business/catalog.json'));
const argv=process.argv.slice(2);
if(argv[0]==='--list'){
 console.log(JSON.stringify(catalog.functions.map(f=>({name:f.name,feature_path:f.feature_path,description:f.description,version:f.version,status:f.status})),null,2));
}else if(argv[0]==='--describe'){
 const entry=catalog.functions.find(f=>f.name===argv[1]);if(!entry)throw Error('Business function not registered');console.log(JSON.stringify(entry,null,2));
}else{
 const state=parse(path.join(root,'任务状态.json'));
 if(!state.execution_allowed)throw Error('Task is waiting for current-round review; execution is not enabled');
 const entry=catalog.functions.find(f=>f.name===argv[0]);if(!entry)throw Error('Business function not registered; discover existing capabilities before adding code');
 const argsIndex=argv.indexOf('--args-file');if(argsIndex<0||!argv[argsIndex+1])throw Error('--args-file is required');
 const argsPath=file(argv[argsIndex+1]),args=parse(argsPath);
 if(!args||typeof args!=='object'||Array.isArray(args))throw Error('Arguments must be an object');
 for(const name of entry.parameters?.required||[])if(!Object.hasOwn(args,name))throw Error('Missing required parameter: '+name);
 if(entry.parameters?.additionalProperties===false)for(const name of Object.keys(args))if(!Object.hasOwn(entry.parameters.properties||{},name))throw Error('Unknown parameter: '+name);
 const env=parse(path.join(root,'环境记录.json')),runtime=parse(path.join(root,'运行上下文.json'));
 if(!env.environment_id)throw Error('Environment baseline is required');
 const implementation=file(entry.file);
 if(!relative(implementation).startsWith('code/business/'))throw Error('Implementation must be in code/business');
 const callId='CALL-'+crypto.randomUUID(),startedAt=new Date().toISOString();
 const snapshots=path.join(root,'审核/代码快照',callId);fs.mkdirSync(snapshots,{recursive:true});
 const argsSnapshot=path.join(snapshots,'arguments.json');fs.copyFileSync(argsPath,argsSnapshot);
 const modules=[...new Set([entry.file,...(entry.modules||[]),'tools/record.cjs','tools/call-business.mjs'])].map(file);
 const records=modules.map(original=>{const rel=relative(original),snapshot=path.join(snapshots,rel+'.txt');fs.mkdirSync(path.dirname(snapshot),{recursive:true});fs.copyFileSync(original,snapshot);return {path:rel,sha256:hash(original),snapshot:relative(snapshot)};});
 const require=createRequire(import.meta.url);
 const {createRecorder}=require('./record.cjs');
 const ctx={taskRoot:root,environmentId:env.environment_id,environment:env,runtime,runId:runtime.extension_run,
  businessCallId:callId,recorder:createRecorder(root,{environment_id:env.environment_id,group:'business',business_call_id:callId})};
 let status='returned',result,error;
 try{const module=await import(pathToFileURL(implementation).href);const fn=module[entry.export]||module.default?.[entry.export];if(typeof fn!=='function')throw Error('Registered export is not a function');result=await fn(ctx,args);}
 catch(cause){status='error';error={type:cause.name,message:String(cause.message)};}
 const endedAt=new Date().toISOString();
 const returnPath=path.join(snapshots,'return.json');fs.writeFileSync(returnPath,JSON.stringify(error?{error}:{observations:result},null,2)+'\n','utf8');
 for(const record of records)record.sha256_after=hash(file(record.path));
 const event={kind:'business_call',call_id:callId,function_name:entry.name,function_version:entry.version,environment_id:env.environment_id,
  started_at:startedAt,ended_at:endedAt,status,args_path:relative(argsSnapshot),args_sha256:hash(argsSnapshot),return_path:relative(returnPath),return_sha256:hash(returnPath),modules:records,...(error?{error}:{})};
 fs.mkdirSync(path.join(root,'运行日志'),{recursive:true});
 fs.appendFileSync(path.join(root,'运行日志/业务调用.jsonl'),JSON.stringify(event)+'\n','utf8');
 // Arguments are kept in their source file; never print private argument values.
 console.log(JSON.stringify({call_id:callId,status,...(error?{error}:{observations:result})},null,2));
 if(status==='error')process.exitCode=1;
}
