import path from 'node:path';
import fs from 'node:fs';
import crypto from 'node:crypto';
import { withApp, sleep } from '../../automation/tpt.mjs';
import { ensureList, openImportDialog, setFileAndConfirm, dialogInfo, skillDirNames } from '../../automation/skills.mjs';

const REG = 'C:\\Users\\Administrator\\.tpt-work\\dsh\\skills-manager\\skill-registry.json';
const SKILLS_DIR = 'C:\\Users\\Administrator\\.tpt-work\\dsh\\skills';
const hash = (p) => crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const regKeys = () => Object.keys(JSON.parse(fs.readFileSync(REG, 'utf8')).entries).sort();

function listFilesRecursive(dir) {
  const out = [];
  if (!fs.existsSync(dir)) return out;
  (function walk(p) {
    for (const x of fs.readdirSync(p)) {
      const fp = path.join(p, x);
      if (fs.statSync(fp).isDirectory()) walk(fp);
      else out.push(path.relative(dir, fp));
    }
  })(dir);
  return out;
}

async function selectOnly(page, file) {
  const f = await ensureList(page);
  await openImportDialog(f);
  const input = f.locator('[role="dialog"] input[type="file"][accept*=".zip"]').first();
  await input.setInputFiles(file);
  await sleep(1500);
  return dialogInfo(f);
}

export async function skillsZip(ctx, args) {
  const run = ctx.runtime.extension_run;
  const zips = path.join(ctx.taskRoot, '夹具', '本轮', run, 'zips');
  const ZIP = (n) => path.join(zips, n);
  return withApp(ctx, async (page) => {
    const { recorder } = ctx;
    const out = {};
    const fileRead = (target, objectId, locator, fn) => recorder.read(target, objectId, { channel: 'file', scope: '正式技能注册表', locator }, fn);

    // ---------- G14-01 select root.zip without confirming ----------
    const id1Before = await fileRead('正式技能身份集合', 'g14.registry.select-root', REG, async () => ({ value: regKeys(), raw: { count: regKeys().length } }));
    const act1 = await recorder.action('选择root.zip不确认', 'setInputFiles', 'zips/root.zip', async () => {
      out.dlg1 = await selectOnly(page, ZIP('root.zip'));
    });
    const sel1 = await recorder.read('确认前已选择的本轮ZIP', 'g14.selected.root', {
      channel: 'dom', scope: '技能iframe/导入弹窗', locator: '[role="dialog"]',
    }, async () => {
      const text = out.dlg1?.text ?? '';
      const value = /root\.zip/.test(text) || (out.dlg1?.after?.text ?? '').includes('root.zip');
      return { value, raw: { dialog: text } };
    });
    const id1After = await fileRead('正式技能身份集合', 'g14.registry.select-root', REG, async () => ({ value: regKeys(), raw: { count: regKeys().length } }));
    const f1 = await ensureList(page);
    await f1.getByRole('button', { name: '取消', exact: true }).first().click({ timeout: 4000 }).catch(() => {});
    await page.keyboard.press('Escape').catch(() => {});

    // ---------- G14-02 confirm root.zip ----------
    const id2Before = await fileRead('正式技能身份集合', 'g14.registry.import-root', REG, async () => ({ value: regKeys(), raw: { count: regKeys().length } }));
    const rootName = 'fast-assert-root-' + run;
    const nestedName = 'fast-assert-nested-' + run;
    const act2 = await recorder.action('确认导入root.zip', 'setInputFiles+click', 'zips/root.zip', async () => {
      const f = await ensureList(page);
      await openImportDialog(f);
      out.imp2 = await setFileAndConfirm(f, ZIP('root.zip'));
    });
    const id2After = await fileRead('正式技能身份集合', 'g14.registry.import-root', REG, async () => ({ value: regKeys(), raw: { count: regKeys().length } }));
    const readDelta = await recorder.read('根ZIP正式对象增量', 'g14.root.delta', {
      channel: 'derived', scope: '正式技能注册表', locator: REG,
    }, async () => {
      const before = id2Before.value.includes(rootName);
      const after = regKeys().includes(rootName);
      const raw = { before_present: before, after_present: after, dir_exists: fs.existsSync(path.join(SKILLS_DIR, rootName)), dialog: (out.imp2?.after?.text ?? '').slice(0, 200) };
      return { value: (after ? 1 : 0) - (before ? 1 : 0), raw, derivation: 'after_present - before_present' };
    });
    const readNested = await recorder.read('内部嵌套技能拆成全局对象', 'g14.nested.count', {
      channel: 'derived', scope: '正式技能注册表', locator: REG,
    }, async () => {
      const keys = regKeys();
      const matches = keys.filter((k) => k.includes('fast-assert-nested-'));
      return { value: matches.length, raw: { matches, dir_exists: fs.existsSync(path.join(SKILLS_DIR, nestedName)) }, derivation: 'registry keys containing fast-assert-nested-' };
    });

    // ---------- G14-03 collection ----------
    const id3Before = await fileRead('正式技能身份集合', 'g14.registry.collection', REG, async () => ({ value: regKeys(), raw: { count: regKeys().length } }));
    const act3 = await recorder.action('确认导入collection.zip', 'setInputFiles+click', 'zips/collection.zip', async () => {
      const f = await ensureList(page);
      await openImportDialog(f);
      out.imp3 = await setFileAndConfirm(f, ZIP('collection.zip'));
    });
    const readCollection = await recorder.read('最终确认后导入的集合内部名列表', 'g14.collection.new', {
      channel: 'derived', scope: '正式技能注册表', locator: REG,
    }, async () => {
      const after = regKeys();
      const beforeSet = new Set(id3Before.value);
      const added = after.filter((k) => !beforeSet.has(k)).sort();
      return { value: added, raw: { added, deep_present: after.some((k) => k.includes('fast-assert-collection-deep-')), dialog: (out.imp3?.after?.text ?? '').slice(0, 200) }, derivation: 'after minus before registry keys, sorted' };
    });

    // ---------- G14-04 trash ----------
    const act4 = await recorder.action('确认导入trash.zip', 'setInputFiles+click', 'zips/trash.zip', async () => {
      const f = await ensureList(page);
      await openImportDialog(f);
      out.imp4 = await setFileAndConfirm(f, ZIP('trash.zip'));
    });
    const readTrash = await recorder.read('安装目录垃圾文件', 'g14.trash.dir', {
      channel: 'file', scope: '本轮垃圾包安装目录', locator: path.join(SKILLS_DIR, 'fast-assert-trash-' + run),
    }, async () => {
      const dir = path.join(SKILLS_DIR, 'fast-assert-trash-' + run);
      const files = listFilesRecursive(dir);
      const junk = files.some((r) => /__MACOSX/.test(r) || /\.DS_Store/.test(r));
      return { value: junk, raw: { dir_exists: fs.existsSync(dir), files }, derivation: '__MACOSX or .DS_Store present in install dir' };
    });

    // ---------- G14-05 same-name skip ----------
    const rootSkill = path.join(SKILLS_DIR, rootName, 'SKILL.md');
    const h5Before = await fileRead('已安装本轮root SKILL.md绝对路径', rootSkill, rootSkill, async () => ({ value: fs.existsSync(rootSkill) ? hash(rootSkill) : '' }));
    const act5 = await recorder.action('重复提交root.zip默认确认', 'setInputFiles+click', 'zips/root.zip', async () => {
      const f = await ensureList(page);
      await openImportDialog(f);
      out.imp5 = await setFileAndConfirm(f, ZIP('root.zip'));
    });
    const readSkip = await recorder.read('最终确认后同名处理结果', 'g14.root.skip', {
      channel: 'dom', scope: '技能iframe/导入弹窗', locator: '[role="dialog"]',
    }, async () => {
      const text = (out.imp5?.after?.text ?? '');
      const skip = /跳过/.test(text);
      return { value: skip ? '跳过' : text.slice(0, 200), raw: { dialog: text } };
    });
    const h5After = await fileRead('已安装本轮root SKILL.md绝对路径', rootSkill, rootSkill, async () => ({ value: fs.existsSync(rootSkill) ? hash(rootSkill) : '' }));

    // ---------- G14-06 mixed ----------
    const id6Before = await fileRead('正式技能身份集合', 'g14.registry.mixed', REG, async () => ({ value: regKeys(), raw: { count: regKeys().length } }));
    const act6 = await recorder.action('确认导入mixed.zip', 'setInputFiles+click', 'zips/mixed.zip', async () => {
      const f = await ensureList(page);
      await openImportDialog(f);
      out.imp6 = await setFileAndConfirm(f, ZIP('mixed.zip'));
    });
    const validName = 'fast-assert-mixed-valid-' + run;
    const invalidName = 'fast-assert-mixed-invalid-' + run;
    const readMixed = await recorder.read('混合导入逐项结果', 'g14.mixed.status', {
      channel: 'derived', scope: '正式技能注册表与导入反馈', locator: REG + ' + dialog',
    }, async () => {
      const text = (out.imp6?.after?.text ?? '') + ' ' + (out.imp6?.before?.text ?? '');
      const after = regKeys();
      const valid = after.includes(validName) ? '成功' : '';
      const rootPresent = after.includes(rootName);
      const rootWas = id6Before.value.includes(rootName);
      const dup = rootPresent && rootWas ? '跳过' : '';
      const invalidPresent = after.includes(invalidName);
      let invalid = '';
      if (!invalidPresent && /invalid|失败|校验|拒绝|缺少|无效|未导入/i.test(text)) invalid = '失败';
      const status = { [validName]: valid, [rootName]: dup, [invalidName]: invalid };
      return { value: status, raw: { dialog: text, invalid_present: invalidPresent }, derivation: 'per-name status from registry presence and dialog text' };
    });
    const readInvalidCount = await recorder.read('混合导入无坏包正式卡', 'g14.mixed.invalid.count', {
      channel: 'derived', scope: '正式技能注册表', locator: REG,
    }, async () => {
      const matches = regKeys().filter((k) => k === invalidName);
      return { value: matches.length, raw: { matches }, derivation: 'registry keys equal to invalid technical name' };
    });

    out.G14_01 = { action: act1.event_id, selected: sel1.event_id, selectedValue: sel1.value, before: id1Before.event_id, after: id1After.event_id };
    out.G14_02 = { action: act2.event_id, deltaRead: readDelta.event_id, delta: readDelta.value, nestedRead: readNested.event_id, nested: readNested.value };
    out.G14_03 = { action: act3.event_id, read: readCollection.event_id, value: readCollection.value };
    out.G14_04 = { action: act4.event_id, read: readTrash.event_id, value: readTrash.value };
    out.G14_05 = { action: act5.event_id, skipRead: readSkip.event_id, skip: readSkip.value, before: h5Before.event_id, after: h5After.event_id, beforeHash: h5Before.value, afterHash: h5After.value };
    out.G14_06 = { action: act6.event_id, read: readMixed.event_id, value: readMixed.value, countRead: readInvalidCount.event_id, count: readInvalidCount.value };
    return out;
  });
}
