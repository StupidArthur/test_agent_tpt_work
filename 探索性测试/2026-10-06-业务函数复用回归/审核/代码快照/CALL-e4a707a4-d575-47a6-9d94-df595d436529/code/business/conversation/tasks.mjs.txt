// 业务函数层：对话与任务（新建、选项目、发送、读取终态）。
// 前置：CDP 实例已启动（环境记录.json）。副作用：UI 操作、可能产生模型调用。
import { withPage, SEL, runningIndicatorCount, waitTerminal } from '../../automation/conversation.mjs';

async function readSessionIdentity(page) {
  return page.evaluate(() => {
    const el = document.querySelector('[data-conversation-session]');
    if (el) return { attr: 'data-conversation-session', value: el.getAttribute('data-conversation-session') };
    const attrs = ['data-session-id', 'data-task-id', 'data-conversation-id', 'data-id'];
    for (const a of attrs) {
      const e = document.querySelector(`[${a}]`);
      if (e) return { attr: a, value: e.getAttribute(a) };
    }
    const active = document.querySelector('[aria-current="page"],[aria-selected="true"]');
    return { attr: null, value: active ? (active.innerText || '').trim() : null };
  });
}

// 新建任务；可选选择已有项目与模型。返回编辑器、模型、项目、会话身份的真实观察。
export async function newTask(ctx, args = {}) {
  const { projectPath = null, model = '标准', reasoning = 'low' } = args;
  return withPage(ctx, async (page) => {
    const rec = ctx.recorder;
    const actionRefs = [];
    const newAct = await rec.action('新建任务', 'click', null, async () => {
      await page.locator(SEL.newTask).first().click();
      await page.waitForTimeout(1800);
    });
    actionRefs.push(newAct.event_id);

    let project = null;
    if (projectPath) {
      const selAct = await rec.action('选择项目', 'open-and-select', { projectPath }, async () => {
        await page.locator(SEL.project).first().click();
        await page.waitForTimeout(800);
        await page.locator('[role="menu"]:visible').getByText(projectPath, { exact: true }).first().click();
        await page.waitForTimeout(1000);
      });
      actionRefs.push(selAct.event_id);
      project = await rec.read('当前任务项目', projectPath, { channel: 'dom', scope: 'composer 项目选择按钮', locator: SEL.project }, async () => {
        const text = (await page.locator(SEL.project).first().innerText()).trim();
        return { value: text, raw: { text }, derivation: 'text=项目按钮实际显示名' };
      });
    }

    const editor = await rec.read('新任务编辑器', 'composer', { channel: 'dom', scope: 'composer 输入区', locator: SEL.composer }, async () => {
      const el = page.locator(SEL.composer).first();
      const count = await el.count();
      const visible = count ? await el.isVisible() : false;
      const editable = count ? await el.getAttribute('contenteditable') : null;
      const raw = { count, visible, editable };
      return { value: visible && editable === 'true', raw, derivation: "visible && contenteditable === 'true'" };
    });

    const modelObs = await rec.read('当前任务模型与推理等级', 'model-trigger', { channel: 'dom', scope: 'composer 模型触发按钮', locator: SEL.model }, async () => {
      const el = page.locator(SEL.model).first();
      const ariaLabel = await el.getAttribute('aria-label');
      const text = (await el.innerText()).trim();
      const m = /当前\s*([^，,]+)/.exec(ariaLabel || '');
      const r = /推理等级\s*([^，,]+)/.exec(ariaLabel || '');
      const raw = { ariaLabel, text };
      return { value: [m ? m[1].trim() : null, r ? r[1].trim() : null], raw };
    });

    const session = await rec.read('本轮任务session身份', 'session', { channel: 'dom', scope: '主界面会话容器', locator: '[data-conversation-session]' }, async () => {
      const s = await readSessionIdentity(page);
      return { value: s.value, raw: s };
    });

    return { action_refs: actionRefs, observations: { project, editor, model: modelObs, session }, args_echo: { model, reasoning } };
  });
}

// 填写并发送消息，等待终态，返回助手最终正文。
export async function sendMessage(ctx, args = {}) {
  const { text, timeoutMs = 90000 } = args;
  if (typeof text !== 'string' || !text.length) throw new Error('text 必填');
  return withPage(ctx, async (page) => {
    const rec = ctx.recorder;
    let preCount = 0;
    const act = await rec.action('composer 发送消息', 'fill-and-send', { chars: text.length }, async () => {
      preCount = await page.locator(SEL.assistantBody).count();
      const c = page.locator(SEL.composer).first();
      await c.click();
      await c.fill(text);
      await page.waitForTimeout(400);
      await page.locator(SEL.send).first().click();
    });
    let wait;
    try { wait = await waitTerminal(page, { timeoutMs, minBubbles: preCount + 1 }); }
    catch (e) { wait = { error: String(e.message), timed_out: true }; }
    const assistant = await rec.read('助手最终正文', 'last-assistant-bubble', { channel: 'dom', scope: '对话 助手气泡', locator: SEL.assistantBody }, async () => {
      const bubbles = page.locator(SEL.assistantBody);
      const n = await bubbles.count();
      const last = n ? (await bubbles.nth(n - 1).innerText()).trim() : null;
      return { value: last, raw: { count: n, last } };
    });
    return { action_refs: [act.event_id], observations: { wait, assistant } };
  });
}

// 读取当前会话终态相关值：运行指示数量、代码块、助手正文。
export async function readTaskState(ctx, args = {}) {
  return withPage(ctx, async (page) => {
    const rec = ctx.recorder;
    const running = await rec.read('可见运行中指示数量', 'running-indicators', { channel: 'dom', scope: '当前会话', locator: 'text=进行中/探索中...' }, async () => {
      const n = await runningIndicatorCount(page);
      return { value: n, raw: { count: n } };
    });
    const assistant = await rec.read('助手最终正文', 'last-assistant-bubble', { channel: 'dom', scope: '对话 助手气泡', locator: SEL.assistantBody }, async () => {
      const bubbles = page.locator(SEL.assistantBody);
      const n = await bubbles.count();
      const last = n ? (await bubbles.nth(n - 1).innerText()).trim() : null;
      return { value: last, raw: { count: n, last } };
    });
    const code = await rec.read('本轮代码块', 'code-block', { channel: 'dom', scope: '助手气泡内 pre/code', locator: `${SEL.assistantBody} pre, ${SEL.assistantBody} code` }, async () => {
      const els = page.locator(`${SEL.assistantBody} pre`).last();
      const n = await els.count();
      let text = n ? await els.textContent() : null;
      if (text != null) text = text.replace(/\r\n/g, '\n').replace(/\r/g, '\n').replace(/\n+$/, '');
      return { value: text, raw: { count: n, text } };
    });
    const fileCards = await rec.read('文件产物卡片数量', 'changed-file-cards', { channel: 'dom', scope: '当前会话对话区 产物卡', locator: '[data-changed-files="true"]' }, async () => {
      const n = await page.locator('[data-changed-files="true"]').count();
      return { value: n, raw: { count: n } };
    });
    return { observations: { running, assistant, code, fileCards } };
  });
}
